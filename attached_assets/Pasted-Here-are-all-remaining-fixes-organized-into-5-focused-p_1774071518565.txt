CREATE TABLE IF NOT EXISTS product_reviews (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  product_id UUID REFERENCES products(id) ON DELETE CASCADE,
  user_id UUID REFERENCES auth.users(id),
  rating INTEGER CHECK (rating >= 1 AND rating <= 5) NOT NULL,
  title TEXT,
  body TEXT NOT NULL CHECK (length(body) >= 20),
  images TEXT[] DEFAULT '{}',
  verified_purchase BOOLEAN DEFAULT false,
  helpful_count INTEGER DEFAULT 0,
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW(),
  UNIQUE(user_id, product_id) -- one review per user per product
);

CREATE TABLE IF NOT EXISTS stock_notifications (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  product_id UUID REFERENCES products(id) ON DELETE CASCADE,
  email TEXT NOT NULL,
  user_id UUID REFERENCES auth.users(id),
  created_at TIMESTAMPTZ DEFAULT NOW(),
  notified_at TIMESTAMPTZ
);

-- Add rating columns to products if not present
ALTER TABLE products 
  ADD COLUMN IF NOT EXISTS avg_rating NUMERIC DEFAULT 0,
  ADD COLUMN IF NOT EXISTS review_count INTEGER DEFAULT 0,
  ADD COLUMN IF NOT EXISTS sale_ends_at TIMESTAMPTZ;

-- RLS for reviews
ALTER TABLE product_reviews ENABLE ROW LEVEL SECURITY;
CREATE POLICY "Public read reviews" ON product_reviews 
  FOR SELECT USING (true);
CREATE POLICY "Auth users insert own review" ON product_reviews 
  FOR INSERT WITH CHECK (auth.uid() = user_id);
CREATE POLICY "Users update own review" ON product_reviews 
  FOR UPDATE USING (auth.uid() = user_id);
CREATE POLICY "Users delete own review" ON product_reviews 
  FOR DELETE USING (auth.uid() = user_id);

-- Auto-update product rating on review insert/update/delete
CREATE OR REPLACE FUNCTION update_product_rating()
RETURNS TRIGGER AS $$
BEGIN
  UPDATE products SET
    avg_rating = (
      SELECT ROUND(AVG(rating)::numeric, 1) 
      FROM product_reviews 
      WHERE product_id = COALESCE(NEW.product_id, OLD.product_id)
    ),
    review_count = (
      SELECT COUNT(*) 
      FROM product_reviews 
      WHERE product_id = COALESCE(NEW.product_id, OLD.product_id)
    )
  WHERE id = COALESCE(NEW.product_id, OLD.product_id);
  RETURN COALESCE(NEW, OLD);
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER refresh_product_rating
  AFTER INSERT OR UPDATE OR DELETE ON product_reviews
  FOR EACH ROW EXECUTE FUNCTION update_product_rating();

export const ReviewsSection = ({ productId }: { productId: string }) => {
  const { user } = useAuth();
  const [sort, setSort] = useState<'recent'|'helpful'|'highest'>('recent');
  const [showForm, setShowForm] = useState(false);

  // Fetch reviews
  const { data: reviews = [], refetch } = useQuery({
    queryKey: ['reviews', productId, sort],
    queryFn: async () => {
      let q = supabase
        .from('product_reviews')
        .select('*, profiles(full_name, avatar_url)')
        .eq('product_id', productId);
      if (sort === 'recent') q = q.order('created_at', { ascending: false });
      if (sort === 'helpful') q = q.order('helpful_count', { ascending: false });
      if (sort === 'highest') q = q.order('rating', { ascending: false });
      const { data } = await q;
      return data || [];
    }
  });

  // Check if user already reviewed
  const { data: userReview } = useQuery({
    queryKey: ['user-review', productId, user?.id],
    queryFn: async () => {
      if (!user) return null;
      const { data } = await supabase
        .from('product_reviews')
        .select('id')
        .eq('product_id', productId)
        .eq('user_id', user.id)
        .maybeSingle();
      return data;
    },
    enabled: !!user
  });

  // Check verified purchase
  const { data: hasPurchased } = useQuery({
    queryKey: ['purchase-check', productId, user?.id],
    queryFn: async () => {
      if (!user) return false;
      const { data } = await supabase
        .from('order_items')
        .select('id')
        .eq('product_id', productId)
        .eq('user_id', user.id)
        .limit(1);
      return (data?.length || 0) > 0;
    },
    enabled: !!user
  });

  // Build rating distribution
  const ratingDist = [5,4,3,2,1].map(star => ({
    star,
    count: reviews.filter((r: any) => r.rating === star).length
  }));
  const avgRating = reviews.length > 0
    ? (reviews.reduce((sum: number, r: any) => sum + r.rating, 0) / reviews.length).toFixed(1)
    : '0.0';

  return (
    <div className="space-y-8">

export const AddReviewForm = ({ productId, onSuccess, onCancel }: any) => {
  const { user } = useAuth();
  const [rating, setRating] = useState(0);
  const [title, setTitle] = useState('');
  const [body, setBody] = useState('');
  const [loading, setLoading] = useState(false);

  const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault();
    if (rating === 0) { toast.error('Please select a star rating'); return; }
    if (body.length < 20) { toast.error('Review must be at least 20 characters'); return; }
    setLoading(true);
    try {
      // Check if user purchased this product for verified badge
      const { data: order } = await supabase
        .from('order_items')
        .select('id')
        .eq('product_id', productId)
        .eq('user_id', user.id)
        .limit(1)
        .maybeSingle();

      const { error } = await supabase.from('product_reviews').insert({
        product_id: productId,
        user_id: user.id,
        rating,
        title: title.trim() || null,
        body: body.trim(),
        verified_purchase: !!order,
      });
      if (error) throw error;
      toast.success('Review posted! Thank you 🎉');
      onSuccess();
    } catch (err: any) {
      if (err.code === '23505') {
        toast.error('You have already reviewed this product');
      } else {
        toast.error('Failed to post review');
      }
    } finally {
      setLoading(false);
    }
  };

  return (
    <form onSubmit={handleSubmit}
          className="p-6 bg-[#C9A84C]/5 rounded-2xl border border-[#C9A84C]/20 space-y-5">
      <h3 className="text-white font-medium">Write Your Review</h3>

      {/* Star selector */}
      <div className="space-y-1.5">
        <label className="text-xs text-gray-400">Your Rating *</label>
        <div className="flex gap-1">
          {[1,2,3,4,5].map(star => (
            <Star
              key={star} size={28}
              className={`cursor-pointer transition-all
                ${star <= rating 
                  ? 'fill-[#C9A84C] text-[#C9A84C] scale-110' 
                  : 'text-gray-600 hover:text-[#C9A84C]'}`}
              onClick={() => setRating(star)}
            />
          ))}
        </div>
      </div>

      {/* Title */}
      <div className="space-y-1.5">
        <label className="text-xs text-gray-400">Review Title (optional)</label>
        <input
          value={title}
          onChange={e => setTitle(e.target.value)}
          placeholder="Summarize your experience"
          maxLength={100}
          className="w-full bg-white/5 border border-white/10 rounded-xl 
                     px-4 py-2.5 text-sm text-white placeholder:text-gray-500
                     focus:border-[#C9A84C]/50 outline-none"

      </div>

      {/* Body */}
      <div className="space-y-1.5">
        <label className="text-xs text-gray-400">
          Your Review * (min 20 characters)
        </label>
        <textarea
          value={body}
          onChange={e => setBody(e.target.value)}
          placeholder="Tell others what you think about this product..."

          className="w-full bg-white/5 border border-white/10 rounded-xl 
                     px-4 py-2.5 text-sm text-white placeholder:text-gray-500
                     focus:border-[#C9A84C]/50 outline-none resize-none"

      </div>

import { useState, useEffect } from "react";
import { Timer } from "lucide-react";

interface CountdownTimerProps {
  expiresAt: string | Date;
  size?: 'sm' | 'md';
}

export const CountdownTimer = ({ expiresAt, size = 'sm' }: CountdownTimerProps) => {
  const [time, setTime] = useState({ h: 0, m: 0, s: 0, expired: false });

  useEffect(() => {
    const tick = () => {
      const diff = new Date(expiresAt).getTime() - Date.now();
      if (diff <= 0) {
        setTime({ h: 0, m: 0, s: 0, expired: true });
        return;
      }
      setTime({
        h: Math.floor(diff / 3600000),
        m: Math.floor((diff % 3600000) / 60000),
        s: Math.floor((diff % 60000) / 1000),
        expired: false
      });
    };
    tick();
    const interval = setInterval(tick, 1000);
    return () => clearInterval(interval);
  }, [expiresAt]);

  if (time.expired) return null;

  const pad = (n: number) => String(n).padStart(2, '0');

  if (size === 'sm') {
    return (
      <div className="flex items-center gap-1 bg-red-500/10 border 
                      border-red-500/20 rounded-full px-2 py-0.5 w-fit">
        <Timer size={10} className="text-red-400" />
        <span className="text-red-400 text-[10px] font-mono">
          {pad(time.h)}:{pad(time.m)}:{pad(time.s)}
        </span>
      </div>
    );
  }

  return (
    <div className="flex items-center gap-2 bg-red-500/10 border 
                    border-red-500/20 rounded-xl px-3 py-2">
      <Timer size={14} className="text-red-400" />
      <span className="text-red-400 text-sm font-medium">Sale ends in</span>
      <div className="flex items-center gap-1 font-mono">
        {[
          { val: time.h, label: 'h' },
          { val: time.m, label: 'm' },
          { val: time.s, label: 's' },
        ].map(({ val, label }, i) => (
          <span key={label} className="flex items-center gap-0.5">
            {i > 0 && <span className="text-red-400/50">:</span>}
            <span className="bg-red-500/20 text-red-300 text-sm px-1.5 py-0.5 
                             rounded font-bold min-w-[28px] text-center">
              {pad(val)}
            </span>
          </span>
        ))}
      </div>
    </div>
  );
};

interface CompareStore {
  items: any[];
  addItem: (product: any) => void;
  removeItem: (id: string) => void;
  clearAll: () => void;
  isInCompare: (id: string) => boolean;
}

export const useCompareStore = create<CompareStore>((set, get) => ({
  items: [],
  addItem: (product) => {
    const { items } = get();
    if (items.length >= 3) {
      // toast is called outside store — handle in component
      return;
    }
    if (!items.find(i => i.id === product.id)) {
      set({ items: [...items, product] });
    }
  },
  removeItem: (id) => set(state => ({ 
    items: state.items.filter(i => i.id !== id) 
  })),
  clearAll: () => set({ items: [] }),
  isInCompare: (id) => get().items.some(i => i.id === id),
}));

const getCompletion = (profile: any, addresses: any[]) => {
  const checks = {
    full_name:  !!profile?.full_name,
    phone:      !!profile?.phone,
    dob:        !!profile?.date_of_birth,
    gender:     !!profile?.gender,
    avatar:     !!profile?.avatar_url,
    address:    addresses?.length > 0,

  const done = Object.values(checks).filter(Boolean).length;

};
