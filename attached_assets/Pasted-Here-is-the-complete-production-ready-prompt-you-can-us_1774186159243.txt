```sql
-- Questions master table
CREATE TABLE questions (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  question_text TEXT NOT NULL,
  question_html TEXT,                    -- rendered HTML for display
  question_type TEXT CHECK (question_type IN ('MCQ','TRUE_FALSE','SHORT_ANSWER','NUMERICAL','CODING')),
  options JSONB,                         -- [{label:'A', text:'...'}, ...]
  correct_answer TEXT NOT NULL,
  explanation TEXT,
  explanation_html TEXT,
  subject TEXT NOT NULL,
  topic TEXT NOT NULL,
  subtopic TEXT,
  difficulty TEXT CHECK (difficulty IN ('EASY','MEDIUM','HARD')),
  exam_type TEXT,                        -- JEE_MAIN, NEET, UPSC, SSC_CGL etc.
  source TEXT CHECK (source IN ('OFFICIAL_PYP','AI_GENERATED','USER_UPLOAD')),
  source_year INTEGER,
  source_paper TEXT,                     -- e.g. "JEE Main 2024 Jan Shift 1"
  marks_positive DECIMAL(4,2) DEFAULT 4,
  marks_negative DECIMAL(4,2) DEFAULT 1,
  has_image BOOLEAN DEFAULT false,
  image_url TEXT,
  latex_present BOOLEAN DEFAULT false,
  uploaded_by UUID REFERENCES auth.users(id),  -- null for official questions
  is_verified BOOLEAN DEFAULT false,
  is_public BOOLEAN DEFAULT true,
  upvotes INTEGER DEFAULT 0,
  downvotes INTEGER DEFAULT 0,
  created_at TIMESTAMPTZ DEFAULT now()
);

-- Exam papers metadata
CREATE TABLE exam_papers (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  exam_name TEXT NOT NULL,               -- "JEE Main"
  exam_type TEXT NOT NULL,               -- "JEE_MAIN"
  year INTEGER NOT NULL,
  session TEXT,                          -- "January", "April"
  shift TEXT,                            -- "Shift 1", "Shift 2"
  paper_number TEXT,                     -- "Paper 1", "Paper 2"
  total_questions INTEGER,
  total_marks INTEGER,
  duration_minutes INTEGER,
  difficulty_level TEXT,
  pdf_url TEXT,                          -- original PDF if available
  created_at TIMESTAMPTZ DEFAULT now()
);

-- Test configurations created by users
CREATE TABLE mock_tests (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID REFERENCES auth.users(id) ON DELETE CASCADE,
  test_name TEXT NOT NULL,
  config JSONB NOT NULL,                 -- full TestConfig object
  question_ids UUID[],                   -- ordered array of selected question IDs
  status TEXT CHECK (status IN ('DRAFT','IN_PROGRESS','COMPLETED','ABANDONED')),
  started_at TIMESTAMPTZ,
  submitted_at TIMESTAMPTZ,
  time_limit_minutes INTEGER,
  created_at TIMESTAMPTZ DEFAULT now()
);

-- User responses during test
CREATE TABLE test_responses (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  test_id UUID REFERENCES mock_tests(id) ON DELETE CASCADE,
  user_id UUID REFERENCES auth.users(id) ON DELETE CASCADE,
  question_id UUID REFERENCES questions(id),
  user_answer TEXT,
  is_correct BOOLEAN,
  is_attempted BOOLEAN DEFAULT false,
  is_marked_review BOOLEAN DEFAULT false,
  time_spent_seconds INTEGER,           -- time on this specific question
  answered_at TIMESTAMPTZ,
  created_at TIMESTAMPTZ DEFAULT now()
);

-- Post-test analysis
CREATE TABLE test_analyses (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  test_id UUID REFERENCES mock_tests(id),
  user_id UUID REFERENCES auth.users(id),
  total_score DECIMAL(8,2),
  max_score DECIMAL(8,2),
  accuracy DECIMAL(5,2),
  attempt_percentage DECIMAL(5,2),
  subject_breakdown JSONB,               -- {Physics: {correct:18, wrong:7, accuracy:72}}
  topic_breakdown JSONB,                 -- {topic: {correct, wrong, time_avg, accuracy}}
  weak_topics TEXT[],
  strong_topics TEXT[],
  time_analysis JSONB,
  ai_analysis_text TEXT,                 -- Claude-generated analysis
  predicted_percentile DECIMAL(5,2),
  improvement_vs_last DECIMAL(5,2),
  created_at TIMESTAMPTZ DEFAULT now()
);

-- Revision list (spaced repetition)
CREATE TABLE revision_list (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID REFERENCES auth.users(id) ON DELETE CASCADE,
  question_id UUID REFERENCES questions(id),
  added_from_test_id UUID REFERENCES mock_tests(id),
  next_review_date DATE DEFAULT CURRENT_DATE + 1,
  review_count INTEGER DEFAULT 0,
  is_mastered BOOLEAN DEFAULT false,
  created_at TIMESTAMPTZ DEFAULT now()
);

-- User topic performance (cumulative across all tests)
CREATE TABLE user_topic_performance (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID REFERENCES auth.users(id) ON DELETE CASCADE,
  topic TEXT NOT NULL,
  subject TEXT NOT NULL,
  exam_type TEXT,
  total_attempted INTEGER DEFAULT 0,
  total_correct INTEGER DEFAULT 0,
  accuracy DECIMAL(5,2) DEFAULT 0,
  avg_time_seconds DECIMAL(6,2),
  last_practiced TIMESTAMPTZ,
  UNIQUE(user_id, topic, exam_type)
);

-- RLS Policies
ALTER TABLE questions ENABLE ROW LEVEL SECURITY;
ALTER TABLE mock_tests ENABLE ROW LEVEL SECURITY;
ALTER TABLE test_responses ENABLE ROW LEVEL SECURITY;
ALTER TABLE test_analyses ENABLE ROW LEVEL SECURITY;
ALTER TABLE revision_list ENABLE ROW LEVEL SECURITY;
ALTER TABLE user_topic_performance ENABLE ROW LEVEL SECURITY;

-- Users see public questions + their own uploads
CREATE POLICY "questions_select" ON questions FOR SELECT 
  USING (is_public = true OR uploaded_by = auth.uid());

-- Users only access their own test data
CREATE POLICY "mock_tests_all" ON mock_tests 
  USING (user_id = auth.uid());
CREATE POLICY "test_responses_all" ON test_responses 
  USING (user_id = auth.uid());
```
